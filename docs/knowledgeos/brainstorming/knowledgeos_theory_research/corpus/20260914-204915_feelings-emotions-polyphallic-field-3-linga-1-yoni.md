## Part 6: The Simulation

### 6.1 The Emotional Simulation

```python
# 3 Linga, 1 Yoni — Emotional Simulation

class Linga:
    def __init__(self, name, desire, confidence, jealousy):
        self.name = name
        self.desire = desire
        self.confidence = confidence
        self.jealousy = jealousy

class Yoni:
    def __init__(self):
        self.overwhelm = 0
        self.curiosity = 0
        self.confusion = 0
        self.tension = 0

    def receive(self, linga):
        self.overwhelm += linga.desire
        self.curiosity += linga.confidence
        self.confusion += linga.jealousy
        self.tension += linga.confidence + linga.jealousy

    def reconcile(self):
        # Reconciliation reduces tension
        self.tension = max(0, self.tension - 5)
        return self.tension == 0
```

### 6.2 The Simulation Run

```python
# Simulation: 3 Linga, 1 Yoni

yoni = Yoni()

linga1 = Linga("L1", desire=8, confidence=9, jealousy=3)
linga2 = Linga("L2", desire=7, confidence=6, jealousy=5)
linga3 = Linga("L3", desire=5, confidence=4, jealousy=7)

yoni.receive(linga1)
yoni.receive(linga2)
yoni.receive(linga3)

print(f"Overwhelm: {yoni.overwhelm}")
print(f"Curiosity: {yoni.curiosity}")
print(f"Confusion: {yoni.confusion}")
print(f"Tension: {yoni.tension}")

reconciled = yoni.reconcile()
print(f"Reconciled: {reconciled}")
```
