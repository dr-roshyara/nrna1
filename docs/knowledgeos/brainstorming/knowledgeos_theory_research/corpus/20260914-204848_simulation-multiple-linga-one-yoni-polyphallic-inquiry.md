## Part 8: The Simulation Code

### 8.1 The Simulation Variables

```python
# Multiple Linga, One Yoni

class Linga:
    def __init__(self, name, proposal, strength):
        self.name = name
        self.proposal = proposal
        self.strength = strength

class Yoni:
    def __init__(self):
        self.proposals = []
        self.total_strength = 0

    def receive(self, linga):
        self.proposals.append(linga)
        self.total_strength += linga.strength

    def reconcile(self):
        # Find the dominant proposal
        max_strength = 0
        dominant = None
        for p in self.proposals:
            if p.strength > max_strength:
                max_strength = p.strength
                dominant = p
        return dominant
```

### 8.2 The Simulation Run

```python
# Simulation: Three Linga, One Yoni

yoni = Yoni()

linga1 = Linga("L1", "Version is 3.69", 3)
linga2 = Linga("L2", "Version is 3.70", 2)
linga3 = Linga("L3", "Version is 3.68", 1)

yoni.receive(linga1)

dominant = yoni.reconcile()

print(f"Dominant proposal: {dominant.proposal}")
print(f"Strength: {dominant.strength}")
print(f"Total proposals: {len(yoni.proposals)}")
```
