### 5.2 The Algorithm

```python
# KnowledgeOS with Multiple Linga and Multiple Yoni

class KnowledgeOS:
    def __init__(self):
        self.proposals = []  # Linga
        self.fields = []     # Yoni
        self.evaluations = {} # Matrix

    def add_proposal(self, proposal):
        self.proposals.append(proposal)

    def add_field(self, field):
        self.fields.append(field)

    def evaluate_all(self):
        for proposal in self.proposals:
            for field in self.fields:
                evaluation = field.evaluate(proposal)
                self.evaluations[(proposal, field)] = evaluation

    def determine_truth(self):
        # Find proposal with highest average evaluation
        scores = {}
        for proposal in self.proposals:
            total = 0
            for field in self.fields:
                total += self.evaluations[(proposal, field)]
            scores[proposal] = total / len(self.fields)
        return max(scores, key=scores.get)
```
